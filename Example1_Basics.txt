program Example1_Basics;

// Déclaration de variables globales (facultatif)
var {
    int g_start = 10;
    string greeting = "Hello";
    int[5] numbers;
}

main {
    // Déclarations locales
    int a = 5;
    float b = 3.5;
    int c; // Initialisée à 0 par défaut (selon EvalVisitor)

    // Opérations arithmétiques
    c = a + (g_start * 2) - 1;

    // Opérateur composé
    a += 10; // a = 15

    // Opérations avec floats
    b = (b * 4.0) / 2.0;

    print(a);      // Attendu : 15
    print(c);      // Attendu : 5 + 20 - 1 = 24
    print(b);      // Attendu : 3.5 * 4.0 / 2.0 = 7.0

    string name = "World";
    print(greeting + " " + name);
    
    numbers[0] = 10;
    numbers[1] = 20;
    numbers[2] = numbers[0] + numbers[1];
    
    print(numbers[2]);
    
    numbers[2] += 5;
    print(numbers[2]);
}
